# portcontext demo — paste ONE line at a time while recording.
# Keep it tight: ~6 commands, no typos. Story = set context once, export everywhere.

# 0. Reset so the recording starts clean (run BEFORE you hit record; don't film this)
Remove-Item -Recurse -Force .portcontext -ErrorAction SilentlyContinue; Clear-Host

# 1. Start fresh
portcontext init --owner "Sajee"

# 2. Teach it who you are and how you work
portcontext add --section identity --text "Sajee - PM at Microsoft, Azure Cosmos DB"
portcontext add --section preferences --text "TypeScript, strict mode, no default exports"

# 3. See what you've captured
portcontext list

# 4. Export to every tool at once (one input -> four files)
portcontext export --to all
